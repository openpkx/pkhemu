cmake_minimum_required(VERSION 3.10)
project(mgba_retro_static C)

set(BUILD_LIBRETRO ON CACHE BOOL "" FORCE)
set(SKIP_LIBRARY ON CACHE BOOL "" FORCE)
set(BUILD_QT OFF CACHE BOOL "" FORCE)
set(BUILD_SDL OFF CACHE BOOL "" FORCE)
set(BUILD_GL OFF CACHE BOOL "" FORCE)
set(BUILD_GLES2 OFF CACHE BOOL "" FORCE)
set(BUILD_GLES3 OFF CACHE BOOL "" FORCE)
set(USE_DEBUGGERS OFF CACHE BOOL "" FORCE)
set(USE_GDB_STUB OFF CACHE BOOL "" FORCE)
set(USE_FFMPEG OFF CACHE BOOL "" FORCE)
set(USE_ZLIB OFF CACHE BOOL "" FORCE)
set(USE_MINIZIP OFF CACHE BOOL "" FORCE)
set(USE_LIBZIP OFF CACHE BOOL "" FORCE)
set(USE_PNG OFF CACHE BOOL "" FORCE)
set(USE_LZMA OFF CACHE BOOL "" FORCE)
set(USE_SQLITE3 OFF CACHE BOOL "" FORCE)
set(USE_ELF OFF CACHE BOOL "" FORCE)
set(USE_LUA OFF CACHE BOOL "" FORCE)
set(USE_JSON_C OFF CACHE BOOL "" FORCE)
set(USE_EDITLINE OFF CACHE BOOL "" FORCE)
set(USE_DISCORD_RPC OFF CACHE BOOL "" FORCE)
set(ENABLE_SCRIPTING OFF CACHE BOOL "" FORCE)
set(BUILD_SUITE OFF CACHE BOOL "" FORCE)
set(BUILD_CINEMA OFF CACHE BOOL "" FORCE)
# States record the core version, so it stays the release number instead of a description of the fork's git checkout.
set(SKIP_GIT ON CACHE BOOL "" FORCE)

add_subdirectory(../vendor/mgba mgba EXCLUDE_FROM_ALL)

# mGBA only builds the libretro core as a shared library; we link it statically on every platform,
# so we rebuild the same target's sources and flags as a static archive.
get_target_property(RETRO_SOURCES mgba_libretro SOURCES)
get_target_property(RETRO_DEFS mgba_libretro COMPILE_DEFINITIONS)
get_target_property(RETRO_INCLUDES mgba_libretro INCLUDE_DIRECTORIES)
get_target_property(RETRO_OPTIONS mgba_libretro COMPILE_OPTIONS)
get_target_property(RETRO_SOURCE_DIR mgba_libretro SOURCE_DIR)
get_target_property(RETRO_BINARY_DIR mgba_libretro BINARY_DIR)
get_directory_property(DIR_DEFS DIRECTORY ${RETRO_SOURCE_DIR} COMPILE_DEFINITIONS)
get_directory_property(DIR_INCLUDES DIRECTORY ${RETRO_SOURCE_DIR} INCLUDE_DIRECTORIES)
get_directory_property(DIR_OPTIONS DIRECTORY ${RETRO_SOURCE_DIR} COMPILE_OPTIONS)

set(ABS_SOURCES)
foreach(src ${RETRO_SOURCES})
	if(IS_ABSOLUTE "${src}")
		list(APPEND ABS_SOURCES "${src}")
	elseif(EXISTS "${RETRO_SOURCE_DIR}/${src}")
		list(APPEND ABS_SOURCES "${RETRO_SOURCE_DIR}/${src}")
	else()
		list(APPEND ABS_SOURCES "${RETRO_BINARY_DIR}/${src}")
	endif()
endforeach()

add_library(mgba_retro STATIC ${ABS_SOURCES})
set_target_properties(mgba_retro PROPERTIES COMPILE_DEFINITIONS "${DIR_DEFS};${RETRO_DEFS}" POSITION_INDEPENDENT_CODE ON)
target_include_directories(mgba_retro PRIVATE ${DIR_INCLUDES} ${RETRO_INCLUDES})
if(DIR_OPTIONS OR RETRO_OPTIONS)
	target_compile_options(mgba_retro PRIVATE ${DIR_OPTIONS} ${RETRO_OPTIONS})
endif()
install(TARGETS mgba_retro ARCHIVE DESTINATION lib)
